    private fun startJ(talk: Boolean = false) {
        pendingTalk = talk
        getSharedPreferences("j", MODE_PRIVATE).edit()
            .putString("gk", gk.text.toString().trim())
            .putString("ek", ek.text.toString().trim())
            .putString("ev", ev.text.toString().trim())
            .apply()
        val need = listOf(
            Manifest.permission.RECORD_AUDIO,
            Manifest.permission.POST_NOTIFICATIONS,
            Manifest.permission.READ_CONTACTS,
            Manifest.permission.CALL_PHONE
        ).filter { checkSelfPermission(it) != PackageManager.PERMISSION_GRANTED }
        if (need.isNotEmpty()) {
            requestPermissions(need.toTypedArray(), 1)
        } else {
            launchService(pendingTalk)
        }
    }

    private fun launchService(talk: Boolean) {
        startForegroundService(Intent(this, JarvisService::class.java).putExtra("talk", talk))
        status.text = if (JarvisAccess.inst == null)
            "Jarvis chal raha hai. Accessibility abhi OFF hai, uske bina app control/search/password nahi chalega."
        else
            "Jarvis chal raha hai. Notification me status dekho."
    }

    override fun onRequestPermissionsResult(code: Int, perms: Array<out String>, res: IntArray) {
        super.onRequestPermissionsResult(code, perms, res)
        if (checkSelfPermission(Manifest.permission.RECORD_AUDIO) == PackageManager.PERMISSION_GRANTED) {
            launchService(pendingTalk)
        } else {
            status.text = "Mic permission chahiye."
        }
    }
}

    override fun onStartCommand(i: Intent?, f: Int, s: Int): Int {

            return START_STICKY

        }
        return START_STICKY

    // Google speech service ki apni mic on/off beep dabane ke liye: sunte waqt

    private var muted = false

    private fun muteBeeps(on: Boolean) {
        if (on == muted) return
        try {
            val am = getSystemService(AUDIO_SERVICE) as AudioManager
            for (s in listOf(AudioManager.STREAM_NOTIFICATION, AudioManager.STREAM_SYSTEM)) {
                am.adjustStreamVolume(s, if (on) AudioManager.ADJUST_MUTE else AudioManager.ADJUST_UNMUTE, 0)
            }
            muted = on
        } catch (_: Exception) {}
    }

    private fun scheduleUnmute() {
        val g = muteGen
        main.postDelayed({ if (g == muteGen) muteBeeps(false) }, 800)
    }

    private fun beepThenCommand() {
        stopSpeech()
        convo = true
        silent = 0
        retries = 0
        lang = "hi-IN"
        if (!isAuthed()) { askCode(); return }
        command(true)
    }

    private fun isAuthed(): Boolean {
        if (authed && System.currentTimeMillis() - lastActive > 30 * 60 * 1000L) authed = false // 30 min kuch nahi bola to dobara lock
        return authed
    }

    private fun normCode(s: String): String {
        var t = s.lowercase()
            .replace("अनिल", "anil").replace("अनील", "anil")
            .replace("twenty seven", "27").replace("twenty-seven", "27").replace("twentyseven", "27")
            .replace("सत्ताईस", "27").replace("सताईस", "27")
            .replace("jarvis", "").replace("code", "").replace("password", "")
        val sb = StringBuilder()
        for (c in t) sb.append(if (c in '०'..'९') ('0' + (c - '०')) else c)
        t = sb.toString()
        return t.filter { it.isLetterOrDigit() }
    }

    private fun codeMatches(ts: List<String>): Boolean {
        val stored = try { Vault.get(this, "__code") } catch (_: Exception) { null } ?: return false
        val want = normCode(stored)
        return want.isNotEmpty() && ts.any { normCode(it) == want }
    }

    private fun askCode() {
        if (System.currentTimeMillis() < lockedUntil) {
            speak("Abhi locked hai. Thodi der baad try karna") { idle() }
            return
        }
        speak("Jarvis online. Code bolo") { command(false) }
    }

    private fun handleCode(ts: List<String>) {
        if (STOP.containsMatchIn(ts[0].lowercase())) { endConvo("Theek hai"); return }
        if (codeMatches(ts)) {
            authed = true
            lastActive = System.currentTimeMillis()
            failCount = 0
            speak("Access granted. Welcome boss", "happy") { command(false) }
        } else {
            failCount++
            if (failCount >= 3) {
                failCount = 0
                lockedUntil = System.currentTimeMillis() + 120000L
                speak("Galat code. Access denied. Do minute baad try karna", "concerned") { idle() }
            } else {
                speak("Galat code. Dobara bolo", "concerned") { command(false) }
            }
        }
    }

    private fun endConvo(msg: String) {
        convo = false
        speak(msg) { idle() }
    }

    // wake mode OFF ho toh mic band rehta hai (BOLO dabane tak)
    private fun idle() {
        if (stopped) return
        convo = false
        if (agentBusy && pending == null) {
            status("Kaam chal raha hai... rokne ke liye ROKO dabao")
            return
        }
        if (prefs.getBoolean("wake", false)) passive()
        else status("Ready. BOLO dabao")
    }

    private fun stopSpeech() {
        speakId++
        tts?.stop()
        try { player?.release() } catch (_: Exception) {}
        player = null
    }

    // stability, style, speed
    private fun voiceCfg(emo: String): Triple<Double, Double, Double> = when (emo) {
        "happy", "excited" -> Triple(0.30, 0.65, 1.08)
        "playful" -> Triple(0.30, 0.55, 1.05)
        "sad", "concerned" -> Triple(0.60, 0.35, 0.92)
        else -> Triple(0.50, 0.30, 1.0)
    }

    private fun speak(text: String, emo: String = "calm", then: () -> Unit) {
        val key = (prefs.getString("ek", "") ?: "").trim()
        if (key.isEmpty()) { speakLocal(text, emo, then); return }
        val mine = ++speakId
        Thread {
            val f = eleven(text, key, emo)
            main.post {
                if (mine != speakId || stopped) return@post
                if (f == null) speakLocal(text, emo, then) else play(f, mine, text, emo, then)
            }
        }.start()
    }

    private fun eleven(text: String, key: String, emo: String): File? {
        return try {
            val vid = (prefs.getString("ev", "") ?: "").trim().ifEmpty { "JBFqnCBsd6RMkjVDRZzb" }
            val c = URL("https://api.elevenlabs.io/v1/text-to-speech/$vid?output_format=mp3_44100_64")
                .openConnection() as HttpURLConnection
            c.requestMethod = "POST"
            c.doOutput = true
            c.connectTimeout = 8000
            c.readTimeout = 20000
            c.setRequestProperty("xi-api-key", key)
            c.setRequestProperty("Content-Type", "application/json")
            val (st, sty, spd) = voiceCfg(emo)
            val body = JSONObject()
                .put("text", text)
                .put("model_id", "eleven_flash_v2_5")
                .put(
                    "voice_settings", JSONObject()
                        .put("stability", st).put("similarity_boost", 0.75)
                        .put("style", sty).put("speed", spd)
                ).toString()
            c.outputStream.use { it.write(body.toByteArray()) }
            if (c.responseCode >= 400) {
                val code = c.responseCode
                main.post { Toast.makeText(this, "ElevenLabs error $code (key/voice ID check karo)", Toast.LENGTH_LONG).show() }
                return null
            }
            cacheDir.listFiles()?.filter { it.name.startsWith("say_") }?.forEach { it.delete() }
            val f = File(cacheDir, "say_" + System.nanoTime() + ".mp3")
            c.inputStream.use { i -> f.outputStream().use { o -> i.copyTo(o) } }
            f
        } catch (_: Exception) { null }
    }

    private fun play(f: File, id: Int, text: String, emo: String, then: () -> Unit) {
        try {
            try { player?.release() } catch (_: Exception) {}
            val mp = MediaPlayer()
            player = mp
            mp.setDataSource(f.absolutePath)
            mp.setOnPreparedListener { it.start() }
            mp.setOnCompletionListener {
                if (id == speakId) {
                    try { mp.release() } catch (_: Exception) {}
                    if (player === mp) player = null
                    then()
                }
            }
            mp.setOnErrorListener { _, _, _ ->
                if (id == speakId) {
                    try { mp.release() } catch (_: Exception) {}
                    if (player === mp) player = null
                    speakLocal(text, emo, then)
                }
                true
            }
            mp.prepareAsync()
        } catch (_: Exception) {
            speakLocal(text, emo, then)
        }
    }

    private fun speakLocal(text: String, emo: String, then: () -> Unit) {
        if (!ttsOk) { then(); return }
        when (emo) {
            "happy", "excited", "playful" -> { tts?.setPitch(0.95f); tts?.setSpeechRate(1.15f) }
            "sad", "concerned" -> { tts?.setPitch(0.8f); tts?.setSpeechRate(0.95f) }
            else -> { tts?.setPitch(0.85f); tts?.setSpeechRate(1.05f) }
        }
        tts?.setOnUtteranceProgressListener(object : UtteranceProgressListener() {
            override fun onStart(id: String?) {}
            override fun onDone(id: String?) { main.post(then) }
            @Deprecated("Deprecated in Java")
            override fun onError(id: String?) { main.post(then) }
        })
        tts?.speak(text, TextToSpeech.QUEUE_FLUSH, null, System.nanoTime().toString())
    }

    private fun listener(onText: (List<String>) -> Unit, onFail: (Int) -> Unit, onReady: () -> Unit = {}) = object : RecognitionListener {
        override fun onResults(b: Bundle?) {
            sid++
            scheduleUnmute()
            val l = b?.getStringArrayList(SpeechRecognizer.RESULTS_RECOGNITION)?.filter { it.isNotBlank() }
            if (l.isNullOrEmpty()) onFail(0) else onText(l)
        }
        override fun onError(e: Int) { sid++; scheduleUnmute(); onFail(e) }
        override fun onReadyForSpeech(p: Bundle?) { onReady() }
        override fun onBeginningOfSpeech() {}
        override fun onRmsChanged(v: Float) {}
        override fun onBufferReceived(b: ByteArray?) {}
        override fun onEndOfSpeech() {}
        override fun onPartialResults(b: Bundle?) {}
        override fun onEvent(t: Int, b: Bundle?) {}
    }

    private fun resetRec() {
        try { rec?.destroy() } catch (_: Exception) {}
        rec = null
    }

    private fun begin(lang: String, l: RecognitionListener, stuck: () -> Unit): Boolean {
        if (!SpeechRecognizer.isRecognitionAvailable(this)) {
            Toast.makeText(this, "Speech service nahi mili. Google app install/update karo.", Toast.LENGTH_LONG).show()
            status("Speech service nahi mili")
            return false
        }
        val r = rec ?: SpeechRecognizer.createSpeechRecognizer(this).also { rec = it }
        try { r.cancel() } catch (_: Exception) {}
        r.setRecognitionListener(l)
        sid++
        val mine = sid
        if (prefs.getBoolean("quiet", true)) { muteGen++; muteBeeps(true) }
        // agar recognizer chupchap atak jaye toh restart
        main.postDelayed({ if (mine == sid && !stopped) { resetRec(); stuck() } }, 25000)
        val i = Intent(RecognizerIntent.ACTION_RECOGNIZE_SPEECH)
            .putExtra(RecognizerIntent.EXTRA_LANGUAGE_MODEL, RecognizerIntent.LANGUAGE_MODEL_FREE_FORM)
            .putExtra(RecognizerIntent.EXTRA_LANGUAGE, lang)
            .putExtra(RecognizerIntent.EXTRA_MAX_RESULTS, 3) // alternatives bhi Gemini ko jaate hain
        if (lang == "en-IN" && !convo) {
            i.putExtra(RecognizerIntent.EXTRA_PREFER_OFFLINE, true)
        } else {
            // bolna khatam hone ke baad thoda zyada ruk ke hi mic band ho (beech me na kate)
            i.putExtra(RecognizerIntent.EXTRA_SPEECH_INPUT_COMPLETE_SILENCE_LENGTH_MILLIS, 1800L)
            i.putExtra(RecognizerIntent.EXTRA_SPEECH_INPUT_POSSIBLY_COMPLETE_SILENCE_LENGTH_MILLIS, 1500L)
            i.putExtra(RecognizerIntent.EXTRA_SPEECH_INPUT_MINIMUM_LENGTH_MILLIS, 2500L)
        }
        r.startListening(i)
        return true
    }

    private fun errName(e: Int) = when (e) {
        1 -> "network timeout"; 2 -> "network"; 3 -> "audio"; 4 -> "server"
        5 -> "client"; 8 -> "busy"; 9 -> "mic permission nahi"; 10 -> "too many requests"
        11 -> "server disconnected"; 12, 13 -> "language"; else -> "code $e"
    }

    // "Jarvis" sunne ka wait (loop)
    private fun passive() {
        if (stopped || agentBusy) return
        status("Sun raha hoon: \"Jarvis\" bolo")
        begin("en-IN", listener(
            onText = { ts ->
                if (WAKE.containsMatchIn(ts.joinToString(" ").lowercase())) {
                    status("Jarvis suna!")
                    beepThenCommand()
                } else passive()
            },
            onFail = { code ->
                if (code != 0 && code != 6 && code != 7) status("Mic: " + errName(code))
                if (code in listOf(3, 5, 8, 9, 11, 12, 13)) resetRec()
                val wait = when (code) { 8, 5, 3 -> 1200L; 10 -> 5000L; else -> 200L }
                main.postDelayed({ passive() }, wait)
            }
        ), stuck = { idle() })
    }

    // wake word / BOLO ke baad asli command. Jawab ke baad mic khud wapas on hota hai.
    private fun command(beepOn: Boolean) {
        if (stopped) return
        convo = true
        status("Mic on ho raha hai...")
        val ok = begin(lang, listener(
            onText = { ts ->
                silent = 0
                retries = 0
                val t = ts[0]
                val low = t.lowercase()
                val p = pending
                if (!isAuthed()) {
                    handleCode(ts)
                } else if (p != null) {
                    pending = null
                    val ok = if (p.code) codeMatches(ts) else (!NO.containsMatchIn(low) && YES.containsMatchIn(low))
                    if (ok) p.yes() else p.no()
                } else if (STOP.containsMatchIn(low)) {
                    endConvo("Theek hai boss")
                } else {
                    think(t, ts.drop(1))
                }
            },
            onFail = { code ->
                when {
                    lang == "hi-IN" && (code == 12 || code == 13) -> { lang = "en-IN"; command(false) }
                    code == 0 || code == 6 || code == 7 -> {
                        silent++
                        if (silent >= 3) {
                            val p = pending
                            pending = null
                            if (p != null) p.no() else endConvo(if (isAuthed()) "Main standby me hoon boss" else "Main standby me hoon")
                        } else main.postDelayed({ command(false) }, 250)
                    }
                    code in listOf(3, 5, 8, 11) && retries < 3 -> {
                        retries++
                        resetRec()
                        main.postDelayed({ command(false) }, 1000)
                    }
                    else -> {
                        status("Mic: " + errName(code))
                        idle()
                    }
                }
            },
            onReady = { status("Boliye boss...") }
        ), stuck = { command(false) })
        if (!ok) convo = false
    }

    private fun turn(role: String, text: String) = JSONObject()
        .put("role", role)
        .put("parts", JSONArray().put(JSONObject().put("text", text)))

    private fun post(model: String, key: String, body: String): String {
        val c = URL("https://generativelanguage.googleapis.com/v1beta/models/$model:generateContent")
            .openConnection() as HttpURLConnection
        c.setRequestProperty("x-goog-api-key", key)
        c.requestMethod = "POST"
        c.doOutput = true
        c.connectTimeout = 8000
        c.readTimeout = 25000
        c.setRequestProperty("Content-Type", "application/json")
        c.outputStream.use { it.write(body.toByteArray()) }
        val stream = if (c.responseCode < 400) c.inputStream else c.errorStream
        return stream.bufferedReader().readText()
    }

    // pehle tez (lite) model, na chale toh normal flash
    private fun generate(key: String, bodyFor: (Boolean) -> String): JSONObject {
        var j = JSONObject()
        val tries = listOf(
            "gemini-flash-lite-latest" to true, "gemini-flash-lite-latest" to false,
            "gemini-flash-latest" to true, "gemini-flash-latest" to false
        )
        for ((m, fast) in tries) {
            j = JSONObject(post(m, key, bodyFor(fast)))
            val code = j.optJSONObject("error")?.optInt("code") ?: 0
            if (code != 404 && code != 400) break
        }
        return j
    }

    private fun textOf(j: JSONObject): String {
        val parts = j.optJSONArray("candidates")?.optJSONObject(0)
            ?.optJSONObject("content")?.optJSONArray("parts") ?: return ""
        val sb = StringBuilder()
        for (k in 0 until parts.length()) sb.append(parts.getJSONObject(k).optString("text"))
        return sb.toString().trim()
    }

    private val appNames by lazy {

        pm.queryIntentActivities(Intent(Intent.ACTION_MAIN).addCategory(Intent.CATEGORY_LAUNCHER), 0)
            .map { it.loadLabel(pm).toString() }.distinct().sorted().joinToString(", ").take(1800)
    }

    private fun memory(): List<String> {

    private fun addMemory(fact: String) {
        if (fact.isBlank()) return
        val l = memory().toMutableList()
        l.add(fact.trim().take(120))
        prefs.edit().putString("mem", JSONArray(l.takeLast(30)).toString()).apply()
    }

    private fun systemPrompt(): String {
        val now = SimpleDateFormat("EEEE, d MMM yyyy, hh:mm a", Locale.ENGLISH).format(Date())
        val mem = memory().joinToString("; ")
        val bm = getSystemService(BATTERY_SERVICE) as BatteryManager
        val bat = bm.getIntProperty(BatteryManager.BATTERY_PROPERTY_CAPACITY)
        return SYSTEM + "\nAbhi ka samay: " + now + ". Battery: " + bat + "%" + (if (bm.isCharging) " (charging)." else ".") +
            (if (mem.isNotEmpty()) "\nUser ke baare me yaad: $mem" else "") +
            "\nPhone me installed apps: " + appNames
    }

    private fun askGemini(q: String, alts: List<String>): String {
        val key = (prefs.getString("gk", "") ?: "").trim()
        val said = if (alts.isEmpty()) q else q + "\n(mic ne ye alternatives bhi suni: " + alts.joinToString(" | ") + ")"
        history.put(turn("user", said))
        try {
            val sys = systemPrompt()
            val j = generate(key) { fast ->
                val gen = JSONObject().put("maxOutputTokens", if (fast) 300 else 600)
                if (fast) gen.put("thinkingConfig", JSONObject().put("thinkingBudget", 0)) // thinking band = jaldi jawab
                JSONObject()
                    .put("systemInstruction", JSONObject().put("parts", JSONArray().put(JSONObject().put("text", sys))))
                    .put("contents", history)
                    .put("generationConfig", gen)
                    .toString()
            }
            if (j.has("error")) {
                history.remove(history.length() - 1)
                val er = j.getJSONObject("error")
                return "Gemini error " + er.optInt("code") + ". " + er.optString("message").take(90)
            }
            val text = textOf(j)
            if (text.isEmpty()) {
                history.remove(history.length() - 1)
                return "Gemini ne khali jawab diya."
            }
            history.put(turn("model", text))
            while (history.length() > 10) { history.remove(0); history.remove(0) }
            return text
        } catch (e: Exception) {
            history.remove(history.length() - 1)
            return "Error. " + e.javaClass.simpleName + " " + (e.message ?: "").take(80)
        }
    }

    private fun respond(a: String) {
        val isErr = a.startsWith("Error") || a.startsWith("Gemini")
        if (isErr) {
            Toast.makeText(this, a.take(200), Toast.LENGTH_LONG).show()
            status(a.take(60))
        }
        val emo = Regex("EMO:\\s*(\\w+)", RegexOption.IGNORE_CASE).find(a)?.groupValues?.get(1)?.lowercase() ?: "calm"
        val m = Regex("ACTION:\\s*(\\{.*\\})", RegexOption.DOT_MATCHES_ALL).find(a)
        var say = a.replace(Regex("ACTION:.*", RegexOption.DOT_MATCHES_ALL), "")
            .replace(Regex("EMO:\\s*\\w+", RegexOption.IGNORE_CASE), "").trim()
        var end = false
        var async = false
        if (m != null) {
            try {
                val r = act(JSONObject(m.groupValues[1]))
                if (r.say != null) say = r.say
                end = r.end
                async = r.async
            } catch (_: Exception) {}
        }
        if (async) { status("Kaam chal raha hai..."); return }
        speak(say.ifBlank { "Ho gaya boss" }.take(300), emo) {
            if (isErr || end) idle()
            else main.postDelayed({ command(false) }, 300) // jawab ke baad mic wapas on
        }
    }

    // contact naam se (ya bola hua number) -> (naam, number)
    private fun resolve(a: JSONObject): Pair<String, String>? {
        val name = a.optString("name").trim()
        val number = a.optString("number").trim()
        if (name.isNotEmpty()) Contacts.find(this, name)?.let { return it }
        if (number.count { it.isDigit() } >= 5) return (name.ifEmpty { number }) to number
        return null
    }

    private fun act(a: JSONObject): Res {
        val type = a.optString("type")
        val needsContact = type in listOf("call", "whatsapp", "sms")
        if (needsContact && a.optString("name").isNotBlank() && !Contacts.has(this))
            return Res("Contacts ki permission chahiye boss. App me allow karo", true)

        when (type) {
            "torch" -> {
                val cm = getSystemService(CAMERA_SERVICE) as CameraManager
                cm.setTorchMode(cm.cameraIdList[0], a.optBoolean("on", true))
            }
            "open_app" -> {
                if (!openApp(a.optString("name"))) return Res(a.optString("name") + " app nahi mili boss")
            }
            "app_search" -> {
                val app = a.optString("app").lowercase()
                val q = a.optString("q")
                val e = Uri.encode(q)
                val link = when {
                    "youtube" in app -> "https://www.youtube.com/results?search_query=$e"
                    "amazon" in app -> "https://www.amazon.in/s?k=$e"
                    "flipkart" in app -> "https://www.flipkart.com/search?q=$e"
                    "play store" in app || "playstore" in app -> "market://search?q=$e"
                    "spotify" in app -> "spotify:search:$e"
                    "maps" in app -> "https://www.google.com/maps/search/$e"
                    else -> null
                }
                if (link != null) go(view(link))
                else {
                    if (JarvisAccess.inst == null) return Res("Accessibility ON karo boss, tab app ke andar search kar paunga", true)
                    if (!openApp(a.optString("app"))) return Res(a.optString("app") + " app nahi mili boss")
                    startAgent("Is app me '$q' search karo aur results dikhao", 3000)
                    return Res(end = true)
                }
            }
            "agent" -> {
                if (JarvisAccess.inst == null) return Res("Pehle Accessibility ON karo boss, uske bina ye nahi hoga", true)
                if (agentBusy) return Res("Abhi ek kaam chal raha hai boss", true)
                val app = a.optString("app").trim()
                if (app.isNotEmpty() && !openApp(app)) return Res("$app app nahi mili boss")
                startAgent(a.optString("goal"), if (app.isEmpty()) 300 else 3000)
                return Res(end = true)
            }
            "call" -> {
                val c = resolve(a) ?: return Res("Contact nahi mila boss")
                val uri = Uri.fromParts("tel", c.second, null)
                val canCall = checkSelfPermission(Manifest.permission.CALL_PHONE) == PackageManager.PERMISSION_GRANTED
                go(Intent(if (canCall) Intent.ACTION_CALL else Intent.ACTION_DIAL, uri))
                return Res(c.first + " ko call laga raha hoon", true)
            }
            "whatsapp" -> {
                val c = resolve(a) ?: return Res("Contact nahi mila boss")
                AutoSend.until = System.currentTimeMillis() + 25000
                go(view("https://wa.me/" + Contacts.waNumber(c.second) + "?text=" + Uri.encode(a.optString("text"))))
                return Res(c.first + " ko WhatsApp kar raha hoon")
            }
            "sms" -> {
                val c = resolve(a) ?: return Res("Contact nahi mila boss")
                go(Intent(Intent.ACTION_SENDTO, Uri.parse("smsto:" + Uri.encode(c.second)))
                    .putExtra("sms_body", a.optString("text")))
                return Res(c.first + " ko message khol diya")
            }
            "email" -> go(
                Intent(Intent.ACTION_SENDTO, Uri.parse("mailto:"))
                    .putExtra(Intent.EXTRA_SUBJECT, a.optString("subject"))
                    .putExtra(Intent.EXTRA_TEXT, a.optString("text"))
            )
            "youtube" -> go(view("https://www.youtube.com/results?search_query=" + Uri.encode(a.optString("q"))))
            "maps" -> go(view("https://www.google.com/maps/search/" + Uri.encode(a.optString("q"))))
            "search" -> go(view("https://www.google.com/search?q=" + Uri.encode(a.optString("q"))))
            "url" -> go(view(a.optString("url")))
            "alarm" -> go(
                Intent(AlarmClock.ACTION_SET_ALARM)
                    .putExtra(AlarmClock.EXTRA_HOUR, a.optInt("hour"))
                    .putExtra(AlarmClock.EXTRA_MINUTES, a.optInt("minute"))
                    .putExtra(AlarmClock.EXTRA_MESSAGE, "Jarvis")
            )
            "timer" -> go(
                Intent(AlarmClock.ACTION_SET_TIMER)
                    .putExtra(AlarmClock.EXTRA_LENGTH, a.optInt("seconds", 60))
                    .putExtra(AlarmClock.EXTRA_MESSAGE, a.optString("text", "Jarvis"))
                    .putExtra(AlarmClock.EXTRA_SKIP_UI, true)
            )
            "volume" -> {
                val am = getSystemService(AUDIO_SERVICE) as AudioManager
                val d = when (a.optString("dir")) {
                    "down" -> AudioManager.ADJUST_LOWER
                    "mute" -> AudioManager.ADJUST_TOGGLE_MUTE
                    else -> AudioManager.ADJUST_RAISE
                }
                am.adjustStreamVolume(AudioManager.STREAM_MUSIC, d, AudioManager.FLAG_SHOW_UI)
            }
            "media" -> {
                val key = when (a.optString("cmd")) {
                    "next" -> KeyEvent.KEYCODE_MEDIA_NEXT
                    "previous" -> KeyEvent.KEYCODE_MEDIA_PREVIOUS
                    else -> KeyEvent.KEYCODE_MEDIA_PLAY_PAUSE
                }
                val am = getSystemService(AUDIO_SERVICE) as AudioManager
                am.dispatchMediaKeyEvent(KeyEvent(KeyEvent.ACTION_DOWN, key))
                am.dispatchMediaKeyEvent(KeyEvent(KeyEvent.ACTION_UP, key))
            }
            "settings" -> go(
                Intent(
                    when (a.optString("page")) {
                        "wifi" -> Settings.ACTION_WIFI_SETTINGS
                        "bluetooth" -> Settings.ACTION_BLUETOOTH_SETTINGS
                        "display" -> Settings.ACTION_DISPLAY_SETTINGS
                        "sound" -> Settings.ACTION_SOUND_SETTINGS
                        "battery" -> Settings.ACTION_BATTERY_SAVER_SETTINGS
                        "location" -> Settings.ACTION_LOCATION_SOURCE_SETTINGS
                        "airplane" -> Settings.ACTION_AIRPLANE_MODE_SETTINGS
                        "apps" -> Settings.ACTION_APPLICATION_SETTINGS
                        else -> Settings.ACTION_SETTINGS
                    }
                )
            )
            "system" -> {
                val acc = JarvisAccess.inst ?: return Res("Accessibility ON karo boss", true)
                val cmd = a.optString("cmd")
                if (!acc.system(cmd)) return Res("Ye nahi ho paya boss")
                if (cmd == "lock") { authed = false; return Res("Screen lock kar diya", true) }
            }
            "jarvis_lock" -> {
                authed = false
                return Res("Jarvis lock ho gaya. Dobara code bolna padega", true)
            }
            "remember" -> addMemory(a.optString("fact"))
            "forget_all" -> prefs.edit().remove("mem").apply()
            "password" -> return passwordAction(a.optString("label"))
            "find_file" -> return findFileAction(a.optString("q"))
            "delete_file" -> return deleteFileAction(a.optString("q"))
            "restore_file" -> {
                val t = lastTrashed ?: return Res("Wapas laane ke liye kuch nahi hai boss")
                val ok = FileOps.restore(t.first, t.second)
                if (ok) lastTrashed = null
                return Res(if (ok) "Wapas rakh di boss" else "Wapas nahi rakh paya boss")
            }
            "empty_trash" -> {

package com.jarvis.app

import android.Manifest
import android.content.Context
import android.content.pm.PackageManager
import android.provider.ContactsContract

/** Phone ke contacts me naam se number dhundhta hai (thoda galat bolne par bhi chalega). */
object Contacts {
    fun has(ctx: Context) =
        ctx.checkSelfPermission(Manifest.permission.READ_CONTACTS) == PackageManager.PERMISSION_GRANTED

    private fun norm(s: String) =
        s.lowercase().replace(Regex("[^\\p{L}\\p{N}]+"), " ").trim()

    private fun lev(a: String, b: String): Int {
        var prev = IntArray(b.length + 1) { it }
        for (i in 1..a.length) {
            val cur = IntArray(b.length + 1)
            cur[0] = i
            for (j in 1..b.length) {
                cur[j] = minOf(
                    prev[j] + 1, cur[j - 1] + 1,
                    prev[j - 1] + if (a[i - 1] == b[j - 1]) 0 else 1
                )
            }
            prev = cur
        }
        return prev[b.length]
    }

    private fun score(q: String, n: String): Int {
        if (n.isEmpty()) return 0
        if (q == n) return 100
        if (n.startsWith(q)) return 90
        if (n.contains(q)) return 80
        val qt = q.split(" ")
        val nt = n.split(" ")
        if (qt.all { t -> nt.any { it.startsWith(t) } }) return 75
        val fuzzy = qt.all { t ->
            t.length >= 3 && nt.any { w -> lev(t, w) <= (if (t.length >= 5) 2 else 1) }
        }
        return if (fuzzy) 60 else 0
    }

    /** return: (saved name, number) ya null */
    fun find(ctx: Context, spoken: String): Pair<String, String>? {
        if (!has(ctx)) return null
        val q = norm(spoken)
        if (q.isEmpty()) return null
        var bestScore = 0
        var best: Pair<String, String>? = null
        ctx.contentResolver.query(
            ContactsContract.CommonDataKinds.Phone.CONTENT_URI,
            arrayOf(
                ContactsContract.CommonDataKinds.Phone.DISPLAY_NAME,
                ContactsContract.CommonDataKinds.Phone.NUMBER
            ), null, null, null
        )?.use { c ->
            while (c.moveToNext()) {
                val name = c.getString(0) ?: continue
                val num = c.getString(1) ?: continue
                val s = score(q, norm(name))
                if (s > bestScore) { bestScore = s; best = name to num }
            }
        }
        return best
    }

    /** WhatsApp (wa.me) ke liye number: sirf digits, 10 digit ho to 91 laga do */
    fun waNumber(raw: String): String {
        val d = raw.filter { it.isDigit() }
        return when {
            raw.trim().startsWith("+") -> d
            d.length == 10 -> "91$d"
            d.length == 11 && d.startsWith("0") -> "91" + d.drop(1)
            else -> d
        }
    }
}

package com.jarvis.app

import android.content.Context
import android.security.keystore.KeyGenParameterSpec
import android.security.keystore.KeyProperties
import android.util.Base64
import org.json.JSONObject
import java.security.KeyStore
import javax.crypto.Cipher
import javax.crypto.KeyGenerator
import javax.crypto.SecretKey
import javax.crypto.spec.GCMParameterSpec

/**
 * Passwords phone ke Android Keystore wali key se encrypt hoke rakhe jaate hain.
 * Ye kabhi Gemini/ElevenLabs ko nahi bheje jaate.
 */
object Vault {
    private const val ALIAS = "jarvis_vault"

    private fun key(): SecretKey {
        val ks = KeyStore.getInstance("AndroidKeyStore").apply { load(null) }
        (ks.getKey(ALIAS, null) as? SecretKey)?.let { return it }
        val kg = KeyGenerator.getInstance(KeyProperties.KEY_ALGORITHM_AES, "AndroidKeyStore")
        kg.init(
            KeyGenParameterSpec.Builder(
                ALIAS, KeyProperties.PURPOSE_ENCRYPT or KeyProperties.PURPOSE_DECRYPT
            )
                .setBlockModes(KeyProperties.BLOCK_MODE_GCM)
                .setEncryptionPaddings(KeyProperties.ENCRYPTION_PADDING_NONE)
                .setKeySize(256)
                .build()
        )
        return kg.generateKey()
    }

    private fun data(ctx: Context): JSONObject =
        try {
            JSONObject(ctx.getSharedPreferences("vault", Context.MODE_PRIVATE).getString("d", "{}") ?: "{}")
        } catch (_: Exception) { JSONObject() }

    private fun save(ctx: Context, j: JSONObject) {
        ctx.getSharedPreferences("vault", Context.MODE_PRIVATE).edit().putString("d", j.toString()).apply()
    }

    fun labels(ctx: Context): List<String> {
        val j = data(ctx)
        return j.keys().asSequence().filter { !it.startsWith("__") }.toList().sorted()
    }

    fun put(ctx: Context, label: String, secret: String) {
        val c = Cipher.getInstance("AES/GCM/NoPadding")
        c.init(Cipher.ENCRYPT_MODE, key())
        val ct = c.doFinal(secret.toByteArray(Charsets.UTF_8))
        val blob = Base64.encodeToString(c.iv + ct, Base64.NO_WRAP)
        val j = data(ctx)
        j.put(label.trim().lowercase(), blob)
        save(ctx, j)
    }

    fun remove(ctx: Context, label: String) {
        val j = data(ctx)
        j.remove(label.trim().lowercase())
        save(ctx, j)
    }

    /** label exact ya milta-julta (jaise "insta" -> "instagram") */
    fun get(ctx: Context, label: String): String? {
        val j = data(ctx)
        val q = label.trim().lowercase()
        if (q.isEmpty()) return null
        val k = j.keys().asSequence().firstOrNull { it == q }
            ?: j.keys().asSequence().firstOrNull { it.contains(q) || q.contains(it) }
            ?: return null
        return try {
            val raw = Base64.decode(j.getString(k), Base64.NO_WRAP)
            val c = Cipher.getInstance("AES/GCM/NoPadding")
            c.init(Cipher.DECRYPT_MODE, key(), GCMParameterSpec(128, raw.copyOfRange(0, 12)))
            String(c.doFinal(raw.copyOfRange(12, raw.size)), Charsets.UTF_8)
        } catch (_: Exception) { null }
    }
}

package com.jarvis.app

import android.os.Build
import android.os.Environment
import java.io.File

/** File dhundhna aur delete (pehle Trash me jaati hai, wapas laa sakte ho). */
object FileOps {
    private val root: File get() = Environment.getExternalStorageDirectory()
    private val trash: File get() = File(root, ".JarvisTrash")

    fun allowed(): Boolean =
        Build.VERSION.SDK_INT >= 30 && Environment.isExternalStorageManager()

    fun search(q: String, limit: Int = 30): List<File> {
        val tokens = q.trim().lowercase().split(" ").filter { it.isNotEmpty() }
        if (tokens.isEmpty()) return emptyList()
        val out = ArrayList<File>()
        try {
            root.walkTopDown()
                .onEnter { d -> d.name != "Android" && !d.name.startsWith(".") }
                .forEach { f ->
                    if (f.isFile) {
                        val n = f.name.lowercase()
                        if (tokens.all { n.contains(it) }) {
                            out.add(f)
                            if (out.size >= limit) return out.sortedByDescending { it.lastModified() }
                        }
                    }
                }
        } catch (_: Exception) {}
        return out.sortedByDescending { it.lastModified() }
    }

    fun toTrash(f: File): File? {
        if (!f.isFile) return null
        trash.mkdirs()
        val dest = File(trash, System.currentTimeMillis().toString() + "_" + f.name)
        return if (f.renameTo(dest)) dest else null
    }

    fun restore(trashed: File, original: File): Boolean {
        if (!trashed.exists() || original.exists()) return false
        original.parentFile?.mkdirs()
        return trashed.renameTo(original)
    }

    fun emptyTrash(): Int {
        var n = 0
        trash.listFiles()?.forEach { if (it.isFile && it.delete()) n++ }
        return n
    }

    fun size(f: File): String {
        val b = f.length()
        return when {
            b > 1_000_000_000 -> "%.1f GB".format(b / 1e9)
            b > 1_000_000 -> "%.1f MB".format(b / 1e6)
            b > 1000 -> "${b / 1000} KB"
            else -> "$b bytes"
        }
    }
}

package com.jarvis.app

import android.accessibilityservice.AccessibilityService
import android.content.Intent
import android.os.Build
import android.os.Bundle
import android.view.accessibility.AccessibilityEvent
import android.view.accessibility.AccessibilityNodeInfo

object AutoSend {
    @Volatile
    var until = 0L
}

class ScreenItem(val node: AccessibilityNodeInfo, val label: String, val kind: String)
class ScreenSnap(val pkg: String, val items: List<ScreenItem>)

class JarvisAccess : AccessibilityService() {

    companion object {
        @Volatile
        var inst: JarvisAccess? = null
    }

    override fun onServiceConnected() {
        inst = this
        super.onServiceConnected()
    }

    override fun onUnbind(intent: Intent?): Boolean {
        inst = null
        return super.onUnbind(intent)
    }

    override fun onAccessibilityEvent(e: AccessibilityEvent?) {
        if (System.currentTimeMillis() > AutoSend.until) return
        val root = rootInActiveWindow ?: return
        val pkg = root.packageName?.toString() ?: return
        if (!pkg.startsWith("com.whatsapp")) return
        val cands = root.findAccessibilityNodeInfosByViewId(pkg + ":id/send") +
            root.findAccessibilityNodeInfosByText("Send")
        val btn = cands.firstOrNull { n ->
            val d = n.contentDescription?.toString() ?: ""
            d.contains("send", true) || d.contains("भेजें")
        } ?: return
        var n: AccessibilityNodeInfo? = btn
        while (n != null && !n.isClickable) n = n.parent
        if (n != null && n.performAction(AccessibilityNodeInfo.ACTION_CLICK)) {
            AutoSend.until = 0L
        }
    }

    override fun onInterrupt() {}

    // ---------- screen padhna aur chalana (agent / password / search ke liye) ----------

    /** Abhi screen par jo dikh raha hai uski list. Password box ka text kabhi nahi padhte. */
    fun snapshot(): ScreenSnap? {
        val root = rootInActiveWindow ?: return null
        val pkg = root.packageName?.toString() ?: ""
        val list = ArrayList<ScreenItem>()
        fun walk(n: AccessibilityNodeInfo?, depth: Int) {
            if (n == null || list.size >= 70 || depth > 25) return
            if (n.isVisibleToUser) {
                val t = (n.text?.toString() ?: "").trim()
                val d = (n.contentDescription?.toString() ?: "").trim()
                val pw = n.isPassword
                val label = if (pw) "" else if (t.isNotEmpty()) t else d
                if (n.isClickable || n.isEditable || label.isNotEmpty()) {
                    val kind = when {
                        pw -> "password"
                        n.isEditable -> "edit"
                        n.isClickable -> "button"
                        else -> "text"
                    }
                    list.add(ScreenItem(n, label.take(50), kind))
                }
            }
            for (i in 0 until n.childCount) walk(n.getChild(i), depth + 1)
        }
        walk(root, 0)
        return ScreenSnap(pkg, list)
    }

    fun click(n: AccessibilityNodeInfo): Boolean {
        try { n.refresh() } catch (_: Exception) {}
        var x: AccessibilityNodeInfo? = n
        while (x != null && !x.isClickable) x = x.parent
        return x?.performAction(AccessibilityNodeInfo.ACTION_CLICK) == true
    }

    fun setText(n: AccessibilityNodeInfo, text: String): Boolean {
        try { n.refresh() } catch (_: Exception) {}
        n.performAction(AccessibilityNodeInfo.ACTION_FOCUS)
        val b = Bundle()
        b.putCharSequence(AccessibilityNodeInfo.ACTION_ARGUMENT_SET_TEXT_CHARSEQUENCE, text)
        return n.performAction(AccessibilityNodeInfo.ACTION_SET_TEXT, b)
    }

    /** keyboard ka Enter/Search (Android 11+) */
    fun enter(n: AccessibilityNodeInfo): Boolean =
        Build.VERSION.SDK_INT >= 30 &&
            n.performAction(AccessibilityNodeInfo.AccessibilityAction.ACTION_IME_ENTER.id)

    fun scroll(): Boolean {
        fun find(n: AccessibilityNodeInfo?): AccessibilityNodeInfo? {
            if (n == null) return null
            if (n.isScrollable) return n
            for (i in 0 until n.childCount) find(n.getChild(i))?.let { return it }
            return null
        }
        val s = find(rootInActiveWindow) ?: return false
        return s.performAction(AccessibilityNodeInfo.ACTION_SCROLL_FORWARD)
    }

    fun back(): Boolean = performGlobalAction(GLOBAL_ACTION_BACK)

    /** home / back / recents / notifications / quick_settings / lock / screenshot */
    fun system(cmd: String): Boolean {
        val action = when (cmd) {
            "home" -> GLOBAL_ACTION_HOME
            "back" -> GLOBAL_ACTION_BACK
            "recents" -> GLOBAL_ACTION_RECENTS
            "notifications" -> GLOBAL_ACTION_NOTIFICATIONS
            "quick_settings" -> GLOBAL_ACTION_QUICK_SETTINGS
            "lock" -> GLOBAL_ACTION_LOCK_SCREEN
            "screenshot" -> GLOBAL_ACTION_TAKE_SCREENSHOT
            else -> -1
        }
        return action != -1 && performGlobalAction(action)
    }

    /** Sirf password wale box me daalta hai (focused ho ya screen par jo pehla mile) */
    fun fillPassword(pw: String): Boolean {
        fun findPw(n: AccessibilityNodeInfo?): AccessibilityNodeInfo? {
            if (n == null) return null
            if (n.isPassword && n.isVisibleToUser) return n
            for (i in 0 until n.childCount) findPw(n.getChild(i))?.let { return it }
            return null
        }
        val root = rootInActiveWindow ?: return false
        val f = root.findFocus(AccessibilityNodeInfo.FOCUS_INPUT)
        val target = if (f != null && f.isPassword) f else findPw(root)
        return target != null && setText(target, pw)
    }
