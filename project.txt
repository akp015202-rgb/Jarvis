package com.jarvis.app

import android.Manifest
import android.app.Activity
import android.content.Intent
import android.content.pm.PackageManager
import android.graphics.Color
import android.os.Bundle
import android.provider.Settings
import android.text.InputType
import android.view.Gravity
import android.widget.Button
import android.widget.EditText
import android.widget.LinearLayout
import android.widget.TextView

class MainActivity : Activity() {
    private lateinit var gk: EditText
    private lateinit var status: TextView

    override fun onCreate(b: Bundle?) {
        super.onCreate(b)
        val p = getSharedPreferences("j", MODE_PRIVATE)
        val cyan = Color.parseColor("#19D3FF")

        fun field(hint: String, k: String) = EditText(this).apply {
            this.hint = hint
            setText(p.getString(k, ""))
            setTextColor(Color.WHITE)
            setHintTextColor(Color.parseColor("#557788"))
            inputType = InputType.TYPE_CLASS_TEXT or InputType.TYPE_TEXT_VARIATION_PASSWORD
            setSingleLine()
        }
        fun btn(t: String, f: () -> Unit) = Button(this).apply {
            text = t
            setTextColor(cyan)
            setBackgroundColor(Color.parseColor("#0C3A4D"))
            setOnClickListener { f() }
        }

        gk = field("Gemini API key", "gk")
        status = TextView(this).apply {
            text = "Gemini key daalo, phir Start dabao."
            setTextColor(Color.parseColor("#7FB7CC"))
            setPadding(0, 32, 0, 0)
        }
        val title = TextView(this).apply {
            text = "JARVIS"
            textSize = 34f
            setTextColor(cyan)
            gravity = Gravity.CENTER
            setPadding(0, 40, 0, 40)
        }

        val root = LinearLayout(this).apply {
            orientation = LinearLayout.VERTICAL
            setBackgroundColor(Color.parseColor("#02070D"))
            setPadding(48, 80, 48, 48)
            addView(title)
            addView(gk)
            addView(btn("Start Jarvis") { startJ() })
            addView(btn("Stop") {
                stopService(Intent(this@MainActivity, JarvisService::class.java))
                status.text = "Jarvis band."
            })
            addView(btn("Battery: No restrictions") {
                startActivity(Intent(Settings.ACTION_IGNORE_BATTERY_OPTIMIZATION_SETTINGS))
            })
            addView(status)
        }
        setContentView(root)
    }

    private fun startJ() {
        getSharedPreferences("j", MODE_PRIVATE).edit()
            .putString("gk", gk.text.toString().trim())
            .apply()
        val need = listOf(Manifest.permission.RECORD_AUDIO, Manifest.permission.POST_NOTIFICATIONS)
            .filter { checkSelfPermission(it) != PackageManager.PERMISSION_GRANTED }
        if (need.isNotEmpty()) {
            requestPermissions(need.toTypedArray(), 1)
        } else {
            launchService()
        }
    }

    private fun launchService() {
        startForegroundService(Intent(this, JarvisService::class.java))
        status.text = "Jarvis chal raha hai. \"Jarvis\" bolo."
    }

    override fun onRequestPermissionsResult(code: Int, perms: Array<out String>, res: IntArray) {
        super.onRequestPermissionsResult(code, perms, res)
        if (checkSelfPermission(Manifest.permission.RECORD_AUDIO) == PackageManager.PERMISSION_GRANTED) {
            launchService()
        } else {
            status.text = "Mic permission chahiye."
        }
    }
}

package com.jarvis.app

import android.app.Notification
import android.app.NotificationChannel
import android.app.NotificationManager
import android.app.PendingIntent
import android.app.Service
import android.content.Intent
import android.content.pm.ServiceInfo
import android.hardware.camera2.CameraManager
import android.net.Uri
import android.os.Bundle
import android.os.Handler
import android.os.IBinder
import android.os.Looper
import android.provider.AlarmClock
import android.speech.RecognitionListener
import android.speech.RecognizerIntent
import android.speech.SpeechRecognizer
import android.speech.tts.TextToSpeech
import android.speech.tts.UtteranceProgressListener
import android.widget.Toast
import org.json.JSONArray
import org.json.JSONObject
import java.net.HttpURLConnection
import java.net.URL
import java.util.Locale

private const val SYSTEM = """Tum Jarvis ho, Iron Man wale assistant jaise. User ko boss bolo.
Roman Hinglish me 1-2 chhote vaakya me jawab do, koi markdown ya emoji nahi.
Agar phone ka kaam ho to jawab ke baad nayi line me likho: ACTION: {json}
Allowed json: {"type":"open_app","name":"YouTube"}, {"type":"call","number":"..."}, {"type":"youtube","q":"..."}, {"type":"maps","q":"..."}, {"type":"url","url":"https://..."}, {"type":"whatsapp","number":"91...","text":"..."}, {"type":"alarm","hour":7,"minute":30}, {"type":"torch","on":true}
Kaam na ho to ACTION line mat likho."""

private val WAKE = Regex("jarvis|jarvish|jaarvis|jarwis|जार्विस|जारविस|जार्विश|जर्विस|जारवीस|जरवीस|जार्वीस")

class JarvisService : Service(), TextToSpeech.OnInitListener {
    private val main = Handler(Looper.getMainLooper())
    private var tts: TextToSpeech? = null
    private var ttsOk = false
    private var rec: SpeechRecognizer? = null
    private var started = false
    private var stopped = false
    private val history = JSONArray()
    private val prefs by lazy { getSharedPreferences("j", MODE_PRIVATE) }
    private val nm by lazy { getSystemService(NotificationManager::class.java) }

    override fun onBind(i: Intent?): IBinder? = null

    override fun onCreate() {
        super.onCreate()
        nm.createNotificationChannel(
            NotificationChannel("jarvis", "Jarvis", NotificationManager.IMPORTANCE_LOW)
        )
        nm.createNotificationChannel(
            NotificationChannel("jarvis_act", "Jarvis actions", NotificationManager.IMPORTANCE_HIGH)
        )
        val n = Notification.Builder(this, "jarvis")
            .setSmallIcon(android.R.drawable.ic_btn_speak_now)
            .setContentTitle("Jarvis ON")
            .setContentText("\"Jarvis\" bolo")
            .setOngoing(true)
            .build()
        startForeground(1, n, ServiceInfo.FOREGROUND_SERVICE_TYPE_MICROPHONE)
        tts = TextToSpeech(this, this)
    }

    override fun onInit(status: Int) {
        ttsOk = status == TextToSpeech.SUCCESS
        if (ttsOk) {
            tts?.language = Locale("en", "IN")
            tts?.setPitch(0.85f)
        }
    }

    override fun onStartCommand(i: Intent?, f: Int, s: Int): Int {
        if (!started) {
            started = true
            main.postDelayed({ passive() }, 1500)
        }
        return START_STICKY
    }

    private fun speak(text: String, then: () -> Unit) {
        if (!ttsOk) { then(); return }
        tts?.setOnUtteranceProgressListener(object : UtteranceProgressListener() {
            override fun onStart(id: String?) {}
            override fun onDone(id: String?) { main.post(then) }
            @Deprecated("Deprecated in Java")
            override fun onError(id: String?) { main.post(then) }
        })
        tts?.speak(text, TextToSpeech.QUEUE_FLUSH, null, System.nanoTime().toString())
    }

    private fun listener(onText: (String) -> Unit, onFail: () -> Unit) = object : RecognitionListener {
        override fun onResults(b: Bundle?) {
            val t = b?.getStringArrayList(SpeechRecognizer.RESULTS_RECOGNITION)?.firstOrNull()
            if (t.isNullOrBlank()) onFail() else onText(t)
        }
        override fun onError(e: Int) { onFail() }
        override fun onReadyForSpeech(p: Bundle?) {}
        override fun onBeginningOfSpeech() {}
        override fun onRmsChanged(v: Float) {}
        override fun onBufferReceived(b: ByteArray?) {}
        override fun onEndOfSpeech() {}
        override fun onPartialResults(b: Bundle?) {}
        override fun onEvent(t: Int, b: Bundle?) {}
    }

    private fun begin(lang: String, l: RecognitionListener): Boolean {
        if (!SpeechRecognizer.isRecognitionAvailable(this)) {
            Toast.makeText(this, "Speech service nahi mili. Google app install karo.", Toast.LENGTH_LONG).show()
            stopSelf()
            return false
        }
        rec?.destroy()
        val r = SpeechRecognizer.createSpeechRecognizer(this)
        rec = r
        r.setRecognitionListener(l)
        r.startListening(
            Intent(RecognizerIntent.ACTION_RECOGNIZE_SPEECH)
                .putExtra(RecognizerIntent.EXTRA_LANGUAGE_MODEL, RecognizerIntent.LANGUAGE_MODEL_FREE_FORM)
                .putExtra(RecognizerIntent.EXTRA_LANGUAGE, lang)
        )
        return true
    }

    // "Jarvis" sunne ka wait (loop)
    private fun passive() {
        if (stopped) return
        begin("en-IN", listener(
            onText = { t ->
                if (WAKE.containsMatchIn(t.lowercase())) {
                    speak("Haan boss") { command() }
                } else passive()
            },
            onFail = { main.postDelayed({ passive() }, 300) }
        ))
    }

    // wake word ke baad asli command
    private fun command() {
        if (stopped) return
        begin("hi-IN", listener(
            onText = { t -> think(t) },
            onFail = { passive() }
        ))
    }

    private fun think(q: String) {
        Thread {
            val reply = askGemini(q)
            main.post { respond(reply) }
        }.start()
    }

    private fun turn(role: String, text: String) = JSONObject()
        .put("role", role)
        .put("parts", JSONArray().put(JSONObject().put("text", text)))

    private fun askGemini(q: String): String {
        val key = prefs.getString("gk", "") ?: ""
        history.put(turn("user", q))
        return try {
            val body = JSONObject()
                .put("systemInstruction", JSONObject().put("parts", JSONArray().put(JSONObject().put("text", SYSTEM))))
                .put("contents", history)
                .put(
                    "generationConfig", JSONObject()
                        .put("maxOutputTokens", 300)
                        .put("thinkingConfig", JSONObject().put("thinkingBudget", 0))
                )
            val c = URL("https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash:generateContent?key=$key")
                .openConnection() as HttpURLConnection
            c.requestMethod = "POST"
            c.doOutput = true
            c.connectTimeout = 15000
            c.readTimeout = 30000
            c.setRequestProperty("Content-Type", "application/json")
            c.outputStream.use { it.write(body.toString().toByteArray()) }
            val stream = if (c.responseCode < 400) c.inputStream else c.errorStream
            val s = stream.bufferedReader().readText()
            val text = JSONObject(s).getJSONArray("candidates").getJSONObject(0)
                .getJSONObject("content").getJSONArray("parts").getJSONObject(0).getString("text")
            history.put(turn("model", text))
            while (history.length() > 10) { history.remove(0); history.remove(0) }
            text
        } catch (e: Exception) {
            history.remove(history.length() - 1)
            "Network ya Gemini key me problem hai boss."
        }
    }

    private fun respond(a: String) {
        val m = Regex("ACTION:\\s*(\\{.*\\})", RegexOption.DOT_MATCHES_ALL).find(a)
        val say = a.replace(Regex("ACTION:.*", RegexOption.DOT_MATCHES_ALL), "").trim()
        speak(say.ifBlank { "Ho gaya boss" }) { passive() }
        if (m != null) {
            try { act(JSONObject(m.groupValues[1])) } catch (_: Exception) {}
        }
    }

    private fun act(a: JSONObject) {
        when (a.optString("type")) {
            "torch" -> {
                val cm = getSystemService(CAMERA_SERVICE) as CameraManager
                cm.setTorchMode(cm.cameraIdList[0], a.optBoolean("on", true))
            }
            "open_app" -> openApp(a.optString("name"))
            "call" -> go(Intent(Intent.ACTION_DIAL, Uri.parse("tel:" + a.optString("number"))))
            "youtube" -> go(view("https://www.youtube.com/results?search_query=" + Uri.encode(a.optString("q"))))
            "maps" -> go(view("https://www.google.com/maps/search/" + Uri.encode(a.optString("q"))))
            "url" -> go(view(a.optString("url")))
            "whatsapp" -> go(
                view(
                    "https://wa.me/" + a.optString("number").filter { it.isDigit() } +
                        "?text=" + Uri.encode(a.optString("text"))
                )
            )
            "alarm" -> go(
                Intent(AlarmClock.ACTION_SET_ALARM)
                    .putExtra(AlarmClock.EXTRA_HOUR, a.optInt("hour"))
                    .putExtra(AlarmClock.EXTRA_MINUTES, a.optInt("minute"))
                    .putExtra(AlarmClock.EXTRA_MESSAGE, "Jarvis")
            )
        }
    }

    private fun view(u: String) = Intent(Intent.ACTION_VIEW, Uri.parse(u))

    private fun openApp(name: String) {
        val pm = packageManager
        val q = Intent(Intent.ACTION_MAIN).addCategory(Intent.CATEGORY_LAUNCHER)
        val hit = pm.queryIntentActivities(q, 0)
            .firstOrNull { it.loadLabel(pm).toString().contains(name, ignoreCase = true) }
        val li = hit?.let { pm.getLaunchIntentForPackage(it.activityInfo.packageName) }
        if (li != null) go(li)
    }

    // Android 10+ background se seedha activity nahi kholne deta,
    // isliye saath me full-screen/tap notification bhi bhejte hain.
    private fun go(i: Intent) {
        i.addFlags(Intent.FLAG_ACTIVITY_NEW_TASK)
        try { startActivity(i) } catch (_: Exception) {}
        val pi = PendingIntent.getActivity(
            this, System.nanoTime().toInt(), i,
            PendingIntent.FLAG_IMMUTABLE or PendingIntent.FLAG_UPDATE_CURRENT
        )
        val n = Notification.Builder(this, "jarvis_act")
            .setSmallIcon(android.R.drawable.ic_btn_speak_now)
            .setContentTitle("Jarvis")
            .setContentText("Kholne ke liye tap karo")
            .setContentIntent(pi)
            .setFullScreenIntent(pi, true)
            .setAutoCancel(true)
            .build()
        nm.notify(2, n)
        main.postDelayed({ nm.cancel(2) }, 10000)
    }

    override fun onDestroy() {
        stopped = true
        main.removeCallbacksAndMessages(null)
        rec?.destroy()
        tts?.shutdown()
        super.onDestroy()
    }
}
