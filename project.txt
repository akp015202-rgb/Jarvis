package com.jarvis.app

import android.Manifest
import android.app.Activity
import android.content.Intent
import android.content.pm.PackageManager
import android.graphics.Color
import android.os.Bundle
import android.provider.Settings
import android.text.InputType
import android.view.Gravity
import android.widget.Button
import android.widget.EditText
import android.widget.LinearLayout
import android.widget.TextView

class MainActivity : Activity() {
    private lateinit var gk: EditText
    private lateinit var status: TextView

    override fun onCreate(b: Bundle?) {
        super.onCreate(b)
        val p = getSharedPreferences("j", MODE_PRIVATE)
        val cyan = Color.parseColor("#19D3FF")

        fun field(hint: String, k: String) = EditText(this).apply {
            this.hint = hint
            setText(p.getString(k, ""))
            setTextColor(Color.WHITE)
            setHintTextColor(Color.parseColor("#557788"))
            inputType = InputType.TYPE_CLASS_TEXT or InputType.TYPE_TEXT_VARIATION_PASSWORD
            setSingleLine()
        }
        fun btn(t: String, f: () -> Unit) = Button(this).apply {
            text = t
            setTextColor(cyan)
            setBackgroundColor(Color.parseColor("#0C3A4D"))
            setOnClickListener { f() }
        }

        gk = field("Gemini API key", "gk")
        status = TextView(this).apply {
            text = "Gemini key daalo, phir Start dabao."
            setTextColor(Color.parseColor("#7FB7CC"))
            setPadding(0, 32, 0, 0)
        }
        val title = TextView(this).apply {
            text = "JARVIS"
            textSize = 34f
            setTextColor(cyan)
            gravity = Gravity.CENTER
            setPadding(0, 40, 0, 40)
        }

        val wakeBtn = btn("") { }
        fun wakeLabel() {
            wakeBtn.text = "Jarvis bolke jagana: " + (if (p.getBoolean("wake", false)) "ON" else "OFF")
        }
        wakeBtn.setOnClickListener {
            p.edit().putBoolean("wake", !p.getBoolean("wake", false)).apply()
            wakeLabel()
            status.text = "Badlav ke baad Start Jarvis dabao."
        }
        wakeLabel()

        val root = LinearLayout(this).apply {
            orientation = LinearLayout.VERTICAL
            setBackgroundColor(Color.parseColor("#02070D"))
            setPadding(48, 80, 48, 48)
            addView(title)
            addView(gk)
            addView(btn("BOLO (ek baar tap, mic on rahega)") { startJ(true) })
            addView(btn("Start Jarvis") { startJ() })
            addView(wakeBtn)
            addView(btn("Stop") {
                stopService(Intent(this@MainActivity, JarvisService::class.java))
                status.text = "Jarvis band."
            })
            addView(btn("Battery: No restrictions") {
                startActivity(Intent(Settings.ACTION_IGNORE_BATTERY_OPTIMIZATION_SETTINGS))
            })
            addView(btn("WhatsApp auto-send: Accessibility ON karo") {
                startActivity(Intent(Settings.ACTION_ACCESSIBILITY_SETTINGS))
            })
            addView(status)
        }
        setContentView(root)
    }

    private var pendingTalk = false

    private fun startJ(talk: Boolean = false) {
        pendingTalk = talk
        getSharedPreferences("j", MODE_PRIVATE).edit()
            .putString("gk", gk.text.toString().trim())
            .apply()
        // contacts + call permission bhi yahin maang lete hain
        val need = listOf(
            Manifest.permission.RECORD_AUDIO,
            Manifest.permission.POST_NOTIFICATIONS,
            Manifest.permission.READ_CONTACTS,
            Manifest.permission.CALL_PHONE
        ).filter { checkSelfPermission(it) != PackageManager.PERMISSION_GRANTED }
        if (need.isNotEmpty()) {
            requestPermissions(need.toTypedArray(), 1)
        } else {
            launchService(pendingTalk)
        }
    }

    private fun launchService(talk: Boolean) {
        startForegroundService(Intent(this, JarvisService::class.java).putExtra("talk", talk))
        status.text = if (checkSelfPermission(Manifest.permission.READ_CONTACTS) == PackageManager.PERMISSION_GRANTED)
            "Jarvis chal raha hai. Notification me status dekho."
        else
            "Jarvis chal raha hai. (Contacts permission nahi di, isliye naam se call/message nahi hoga)"
    }

    override fun onRequestPermissionsResult(code: Int, perms: Array<out String>, res: IntArray) {
        super.onRequestPermissionsResult(code, perms, res)
        if (checkSelfPermission(Manifest.permission.RECORD_AUDIO) == PackageManager.PERMISSION_GRANTED) {
            launchService(pendingTalk)
        } else {
            status.text = "Mic permission chahiye."
        }
    }
}

private const val SYSTEM = """Tum Jarvis ho, Iron Man wale assistant jaise. User ko boss bolo.
Roman Hinglish me 1 chhota vaakya me jawab do (bahut short), koi markdown ya emoji nahi.
Agar phone ka kaam ho to jawab ke baad nayi line me likho: ACTION: {json}
Allowed json:
{"type":"open_app","name":"YouTube"} (phone ka koi bhi app, naam jaisa app ke icon ke neeche dikhta hai)
{"type":"call","name":"contact ka naam"}
{"type":"whatsapp","name":"contact ka naam","text":"..."}
{"type":"sms","name":"contact ka naam","text":"..."}
{"type":"email","subject":"...","text":"..."}
{"type":"youtube","q":"..."}, {"type":"maps","q":"..."}, {"type":"search","q":"..."}, {"type":"url","url":"https://..."}
{"type":"alarm","hour":7,"minute":30}, {"type":"timer","seconds":300}
{"type":"torch","on":true}
{"type":"volume","dir":"up"} (up/down/mute)
{"type":"media","cmd":"play_pause"} (play_pause/next/previous)
{"type":"settings","page":"wifi"} (wifi/bluetooth/display/sound/battery/location/airplane/apps/main)
Contact ka naam waisa hi likho jaisa user ne bola, Roman letters me. "number" key sirf tab do jab user ne khud number bola ho (call/whatsapp/sms me "number":"..." bhi chalta hai).
Kaam na ho to ACTION line mat likho."""

private val WAKE = Regex("jarvis|jarvish|jaarvis|jarwis|जार्विस|जारविस|जार्विश|जर्विस|जारवीस|जरवीस|जार्वीस")
private val STOP = Regex("^\\s*(band|bas|stop|bye|ruk|ruko|exit|chup)\\b")

private class Res(val say: String? = null, val end: Boolean = false)

class JarvisService : Service(), TextToSpeech.OnInitListener {
    private val main = Handler(Looper.getMainLooper())
    private var tts: TextToSpeech? = null
    private var ttsOk = false
    private var rec: SpeechRecognizer? = null
    private var started = false
    private var stopped = false
    private var sid = 0
    private val history = JSONArray()
    private val prefs by lazy { getSharedPreferences("j", MODE_PRIVATE) }
    private val nm by lazy { getSystemService(NotificationManager::class.java) }

    // conversation mode: BOLO ek baar dabao, "band/stop" bolne ya 3 baar chup rehne tak mic on
    private var convo = false
    private var silent = 0
    private var retries = 0
    private var lang = "hi-IN"

    override fun onBind(i: Intent?): IBinder? = null

    private fun note(text: String): Notification {
        val talkPi = PendingIntent.getService(
            this, 7, Intent(this, JarvisService::class.java).putExtra("talk", true),
            PendingIntent.FLAG_IMMUTABLE or PendingIntent.FLAG_UPDATE_CURRENT
        )
        val act = Notification.Action.Builder(
            android.graphics.drawable.Icon.createWithResource(this, android.R.drawable.ic_btn_speak_now),
            "BOLO", talkPi
        ).build()
        return Notification.Builder(this, "jarvis")
            .setSmallIcon(android.R.drawable.ic_btn_speak_now)
            .setContentTitle("Jarvis ON")
            .setContentText(text)
            .setOnlyAlertOnce(true)
            .setOngoing(true)
            .addAction(act)
            .build()
    }

    private fun status(text: String) {
        nm.notify(1, note(text))
    }

    override fun onCreate() {
        super.onCreate()
        nm.createNotificationChannel(
            NotificationChannel("jarvis", "Jarvis", NotificationManager.IMPORTANCE_LOW)
        )
        nm.createNotificationChannel(
            NotificationChannel("jarvis_act", "Jarvis actions", NotificationManager.IMPORTANCE_HIGH)
        )
        startForeground(1, note("Start ho raha hai..."), ServiceInfo.FOREGROUND_SERVICE_TYPE_MICROPHONE)
        tts = TextToSpeech(this, this)
    }

    override fun onInit(s: Int) {
        ttsOk = s == TextToSpeech.SUCCESS
        if (ttsOk) {
            tts?.language = Locale("en", "IN")
            tts?.setPitch(0.85f)
            tts?.setSpeechRate(1.1f)
        }
    }

    override fun onStartCommand(i: Intent?, f: Int, s: Int): Int {
        val talk = i?.getBooleanExtra("talk", false) == true
        if (!started) {
            started = true
            main.postDelayed({ if (talk) beepThenCommand() else idle() }, 800)
        } else if (talk) {
            main.post { beepThenCommand() }
        } else {
            main.post { idle() }
        }
        return START_STICKY
    }

    private fun beep() {
        try {
            val tg = ToneGenerator(AudioManager.STREAM_MUSIC, 80)
            tg.startTone(ToneGenerator.TONE_PROP_ACK, 120)
            main.postDelayed({ tg.release() }, 400)
        } catch (_: Exception) {}
    }

    private fun beepThenCommand() {
        tts?.stop()
        convo = true
        silent = 0
        retries = 0
        lang = "hi-IN"
        command(true)
    }

    private fun endConvo(msg: String) {
        convo = false
        speak(msg) { idle() }
    }

    // wake mode OFF ho toh mic band rehta hai (BOLO dabane tak)
    private fun idle() {
        if (stopped) return
        convo = false
        if (prefs.getBoolean("wake", false)) passive()
        else status("Ready. BOLO dabao")
    }

    private fun speak(text: String, then: () -> Unit) {
        if (!ttsOk) { then(); return }
        tts?.setOnUtteranceProgressListener(object : UtteranceProgressListener() {
            override fun onStart(id: String?) {}
            override fun onDone(id: String?) { main.post(then) }
            @Deprecated("Deprecated in Java")
            override fun onError(id: String?) { main.post(then) }
        })
        tts?.speak(text, TextToSpeech.QUEUE_FLUSH, null, System.nanoTime().toString())
    }

    private fun listener(onText: (String) -> Unit, onFail: (Int) -> Unit, onReady: () -> Unit = {}) = object : RecognitionListener {
        override fun onResults(b: Bundle?) {
            sid++
            val t = b?.getStringArrayList(SpeechRecognizer.RESULTS_RECOGNITION)?.firstOrNull()
            if (t.isNullOrBlank()) onFail(0) else onText(t)
        }
        override fun onError(e: Int) { sid++; onFail(e) }
        override fun onReadyForSpeech(p: Bundle?) { onReady() }
        override fun onBeginningOfSpeech() {}
        override fun onRmsChanged(v: Float) {}
        override fun onBufferReceived(b: ByteArray?) {}
        override fun onEndOfSpeech() {}
        override fun onPartialResults(b: Bundle?) {}
        override fun onEvent(t: Int, b: Bundle?) {}
    }

    private fun resetRec() {
        try { rec?.destroy() } catch (_: Exception) {}
        rec = null
    }

    private fun begin(lang: String, l: RecognitionListener, stuck: () -> Unit): Boolean {
        if (!SpeechRecognizer.isRecognitionAvailable(this)) {
            Toast.makeText(this, "Speech service nahi mili. Google app install/update karo.", Toast.LENGTH_LONG).show()
            status("Speech service nahi mili")
            return false
        }
        val r = rec ?: SpeechRecognizer.createSpeechRecognizer(this).also { rec = it }
        try { r.cancel() } catch (_: Exception) {}
        r.setRecognitionListener(l)
        sid++
        val mine = sid
        // agar recognizer chupchap atak jaye toh restart
        main.postDelayed({ if (mine == sid && !stopped) { resetRec(); stuck() } }, 25000)
        val i = Intent(RecognizerIntent.ACTION_RECOGNIZE_SPEECH)
            .putExtra(RecognizerIntent.EXTRA_LANGUAGE_MODEL, RecognizerIntent.LANGUAGE_MODEL_FREE_FORM)
            .putExtra(RecognizerIntent.EXTRA_LANGUAGE, lang)
        if (lang == "en-IN" && !convo) {
            i.putExtra(RecognizerIntent.EXTRA_PREFER_OFFLINE, true)
        } else {
            // bolna khatam hone ke baad thoda zyada ruk ke hi mic band ho (beech me na kate)
            i.putExtra(RecognizerIntent.EXTRA_SPEECH_INPUT_COMPLETE_SILENCE_LENGTH_MILLIS, 1800L)
            i.putExtra(RecognizerIntent.EXTRA_SPEECH_INPUT_POSSIBLY_COMPLETE_SILENCE_LENGTH_MILLIS, 1500L)
            i.putExtra(RecognizerIntent.EXTRA_SPEECH_INPUT_MINIMUM_LENGTH_MILLIS, 2500L)
        }
        r.startListening(i)
        return true
    }

    private fun errName(e: Int) = when (e) {
        1 -> "network timeout"; 2 -> "network"; 3 -> "audio"; 4 -> "server"
        5 -> "client"; 8 -> "busy"; 9 -> "mic permission nahi"; 10 -> "too many requests"
        11 -> "server disconnected"; 12, 13 -> "language"; else -> "code $e"
    }

    // "Jarvis" sunne ka wait (loop)
    private fun passive() {
        if (stopped) return
        status("Sun raha hoon: \"Jarvis\" bolo")
        begin("en-IN", listener(
            onText = { t ->
                if (WAKE.containsMatchIn(t.lowercase())) {
                    status("Jarvis suna!")
                    beepThenCommand()
                } else passive()
            },
            onFail = { code ->
                if (code != 0 && code != 6 && code != 7) status("Mic: " + errName(code))
                if (code in listOf(3, 5, 8, 9, 11, 12, 13)) resetRec()
                val wait = when (code) { 8, 5, 3 -> 1200L; 10 -> 5000L; else -> 200L }
                main.postDelayed({ passive() }, wait)
            }
        ), stuck = { idle() })
    }

    // wake word / BOLO ke baad asli command. Jawab ke baad mic khud wapas on hota hai.
    private fun command(beepOn: Boolean) {
        if (stopped) return
        convo = true
        status("Mic on ho raha hai...")
        val ok = begin(lang, listener(
            onText = { t ->
                silent = 0
                retries = 0
                if (STOP.containsMatchIn(t.lowercase())) endConvo("Theek hai boss")
                else think(t)
            },
            onFail = { code ->
                when {
                    lang == "hi-IN" && (code == 12 || code == 13) -> { lang = "en-IN"; command(false) }
                    code == 0 || code == 6 || code == 7 -> {
                        silent++
                        if (silent >= 3) endConvo("Main standby me hoon boss")
                        else main.postDelayed({ command(false) }, 250)
                    }
                    code in listOf(3, 5, 8, 11) && retries < 3 -> {
                        retries++
                        resetRec()
                        main.postDelayed({ command(false) }, 1000)
                    }
                    else -> {
                        status("Mic: " + errName(code))
                        idle()
                    }
                }
            },
            onReady = { if (beepOn) beep(); status("Boliye boss...") }
        ), stuck = { command(false) })
        if (!ok) convo = false
    }

    private fun think(q: String) {
        status("Soch raha hoon: " + q.take(40))
        Thread {
            val reply = askGemini(q)
            main.post { respond(reply) }
        }.start()
    }

    private fun turn(role: String, text: String) = JSONObject()
        .put("role", role)
        .put("parts", JSONArray().put(JSONObject().put("text", text)))

    private fun post(model: String, key: String, body: String): String {
        val c = URL("https://generativelanguage.googleapis.com/v1beta/models/$model:generateContent")
            .openConnection() as HttpURLConnection
        c.setRequestProperty("x-goog-api-key", key)
        c.requestMethod = "POST"
        c.doOutput = true
        c.connectTimeout = 8000
        c.readTimeout = 25000
        c.setRequestProperty("Content-Type", "application/json")
        c.outputStream.use { it.write(body.toByteArray()) }
        val stream = if (c.responseCode < 400) c.inputStream else c.errorStream
        return stream.bufferedReader().readText()
    }

    private fun buildBody(noThinking: Boolean): String {
        val gen = JSONObject().put("maxOutputTokens", if (noThinking) 220 else 512)
        // thinking band = jawab jaldi aata hai
        if (noThinking) gen.put("thinkingConfig", JSONObject().put("thinkingBudget", 0))
        return JSONObject()
            .put("systemInstruction", JSONObject().put("parts", JSONArray().put(JSONObject().put("text", SYSTEM))))
            .put("contents", history)
            .put("generationConfig", gen)
            .toString()
    }

    private fun askGemini(q: String): String {
        val key = (prefs.getString("gk", "") ?: "").trim()
        history.put(turn("user", q))
        try {
            var s = ""
            var j = JSONObject()
            // pehle tez (lite) model, na chale toh normal flash
            val tries = listOf(
                "gemini-flash-lite-latest" to true, "gemini-flash-lite-latest" to false,
                "gemini-flash-latest" to true, "gemini-flash-latest" to false
            )
            for ((m, fast) in tries) {
                s = post(m, key, buildBody(fast))
                j = JSONObject(s)
                val code = j.optJSONObject("error")?.optInt("code") ?: 0
                if (code != 404 && code != 400) break
            }
            if (j.has("error")) {
                history.remove(history.length() - 1)
                val er = j.getJSONObject("error")
                return "Gemini error " + er.optInt("code") + ". " + er.optString("message").take(90)
            }
            val cand = j.optJSONArray("candidates")?.optJSONObject(0)
            val parts = cand?.optJSONObject("content")?.optJSONArray("parts")
            val sb = StringBuilder()
            if (parts != null) {
                for (k in 0 until parts.length()) sb.append(parts.getJSONObject(k).optString("text"))
            }
            val text = sb.toString().trim()
            if (text.isEmpty()) {
                history.remove(history.length() - 1)
                return "Gemini ne khali jawab diya. " + (cand?.optString("finishReason") ?: s.take(80))
            }
            history.put(turn("model", text))
            while (history.length() > 10) { history.remove(0); history.remove(0) }
            return text
        } catch (e: Exception) {
            history.remove(history.length() - 1)
            return "Error. " + e.javaClass.simpleName + " " + (e.message ?: "").take(80)
        }
    }

    private fun respond(a: String) {
        val isErr = a.startsWith("Error") || a.startsWith("Gemini")
        if (isErr) {
            Toast.makeText(this, a.take(200), Toast.LENGTH_LONG).show()
            status(a.take(60))
        }
        val m = Regex("ACTION:\\s*(\\{.*\\})", RegexOption.DOT_MATCHES_ALL).find(a)
        var say = a.replace(Regex("ACTION:.*", RegexOption.DOT_MATCHES_ALL), "").trim()
        var end = false
        if (m != null) {
            try {
                val r = act(JSONObject(m.groupValues[1]))
                if (r.say != null) say = r.say
                end = r.end
            } catch (_: Exception) {}
        }
        speak(say.ifBlank { "Ho gaya boss" }.take(300)) {
            if (isErr || end) idle()
            else main.postDelayed({ command(false) }, 300) // jawab ke baad mic wapas on
        }
    }

    // contact naam se (ya bola hua number) -> (naam, number)
    private fun resolve(a: JSONObject): Pair<String, String>? {
        val name = a.optString("name").trim()
        val number = a.optString("number").trim()
        if (name.isNotEmpty()) Contacts.find(this, name)?.let { return it }
        if (number.count { it.isDigit() } >= 5) return (name.ifEmpty { number }) to number
        return null
    }

    private fun act(a: JSONObject): Res {
        val type = a.optString("type")
        val needsContact = type in listOf("call", "whatsapp", "sms")
        if (needsContact && a.optString("name").isNotBlank() && !Contacts.has(this))
            return Res("Contacts ki permission chahiye boss. App me allow karo", true)

        when (type) {
            "torch" -> {
                val cm = getSystemService(CAMERA_SERVICE) as CameraManager
                cm.setTorchMode(cm.cameraIdList[0], a.optBoolean("on", true))
            }
            "open_app" -> {
                if (!openApp(a.optString("name"))) return Res(a.optString("name") + " app nahi mili boss")
            }
            "call" -> {
                val c = resolve(a) ?: return Res("Contact nahi mila boss")
                val uri = Uri.fromParts("tel", c.second, null)

    private fun openApp(name: String): Boolean {
        val n = name.trim().lowercase()
        if (n.isEmpty()) return false
        val pm = packageManager
        val q = Intent(Intent.ACTION_MAIN).addCategory(Intent.CATEGORY_LAUNCHER)
        var best: String? = null
        var bestScore = 0
        for (ri in pm.queryIntentActivities(q, 0)) {
            val l = ri.loadLabel(pm).toString().lowercase()
            val s = when {
                l == n -> 4
                l.startsWith(n) -> 3
                l.contains(n) -> 2
                l.length >= 3 && n.contains(l) -> 1
                else -> 0
            }
            if (s > bestScore) { bestScore = s; best = ri.activityInfo.packageName }
        }
        val li = best?.let { pm.getLaunchIntentForPackage(it) } ?: return false
        go(li)
        return true
    }

    // Android 10+ background se seedha activity nahi kholne deta,
    // isliye saath me full-screen/tap notification bhi bhejte hain.
    private fun go(i: Intent) {
        i.addFlags(Intent.FLAG_ACTIVITY_NEW_TASK)
        try { startActivity(i) } catch (_: Exception) {}
        val pi = PendingIntent.getActivity(
            this, System.nanoTime().toInt(), i,
            PendingIntent.FLAG_IMMUTABLE or PendingIntent.FLAG_UPDATE_CURRENT
        )
        val n = Notification.Builder(this, "jarvis_act")
            .setSmallIcon(android.R.drawable.ic_btn_speak_now)
            .setContentTitle("Jarvis")
            .setContentText("Kholne ke liye tap karo")
            .setContentIntent(pi)
            .setFullScreenIntent(pi, true)
            .setAutoCancel(true)
            .build()
        nm.notify(2, n)
        main.postDelayed({ nm.cancel(2) }, 10000)
    }

    override fun onDestroy() {
        stopped = true
        main.removeCallbacksAndMessages(null)
        resetRec()
        tts?.shutdown()
        super.onDestroy()
    }

package com.jarvis.app

import android.Manifest
import android.content.Context
import android.content.pm.PackageManager
import android.provider.ContactsContract

/** Phone ke contacts me naam se number dhundhta hai (thoda galat bolne par bhi chalega). */
object Contacts {
    fun has(ctx: Context) =
        ctx.checkSelfPermission(Manifest.permission.READ_CONTACTS) == PackageManager.PERMISSION_GRANTED

    private fun norm(s: String) =
        s.lowercase().replace(Regex("[^\\p{L}\\p{N}]+"), " ").trim()

    private fun lev(a: String, b: String): Int {
        var prev = IntArray(b.length + 1) { it }
        for (i in 1..a.length) {
            val cur = IntArray(b.length + 1)
            cur[0] = i
            for (j in 1..b.length) {
                cur[j] = minOf(
                    prev[j] + 1, cur[j - 1] + 1,
                    prev[j - 1] + if (a[i - 1] == b[j - 1]) 0 else 1
                )
            }
            prev = cur
        }
        return prev[b.length]
    }

    private fun score(q: String, n: String): Int {
        if (n.isEmpty()) return 0
        if (q == n) return 100
        if (n.startsWith(q)) return 90
        if (n.contains(q)) return 80
        val qt = q.split(" ")
        val nt = n.split(" ")
        if (qt.all { t -> nt.any { it.startsWith(t) } }) return 75
        val fuzzy = qt.all { t ->
            t.length >= 3 && nt.any { w -> lev(t, w) <= (if (t.length >= 5) 2 else 1) }
        }
        return if (fuzzy) 60 else 0
    }

    /** return: (saved name, number) ya null */
    fun find(ctx: Context, spoken: String): Pair<String, String>? {
        if (!has(ctx)) return null
        val q = norm(spoken)
        if (q.isEmpty()) return null
        var bestScore = 0
        var best: Pair<String, String>? = null
        ctx.contentResolver.query(
            ContactsContract.CommonDataKinds.Phone.CONTENT_URI,
            arrayOf(
                ContactsContract.CommonDataKinds.Phone.DISPLAY_NAME,
                ContactsContract.CommonDataKinds.Phone.NUMBER
            ), null, null, null
        )?.use { c ->
            while (c.moveToNext()) {
                val name = c.getString(0) ?: continue
                val num = c.getString(1) ?: continue
                val s = score(q, norm(name))
                if (s > bestScore) { bestScore = s; best = name to num }
            }
        }
        return best
    }

    /** WhatsApp (wa.me) ke liye number: sirf digits, 10 digit ho to 91 laga do */
    fun waNumber(raw: String): String {
        val d = raw.filter { it.isDigit() }
        return when {
            raw.trim().startsWith("+") -> d
            d.length == 10 -> "91$d"
            d.length == 11 && d.startsWith("0") -> "91" + d.drop(1)
            else -> d
        }
    }
}

package com.jarvis.app

import android.accessibilityservice.AccessibilityService
import android.view.accessibility.AccessibilityEvent
import android.view.accessibility.AccessibilityNodeInfo

object AutoSend {
    @Volatile
    var until = 0L
}

class JarvisAccess : AccessibilityService() {
    override fun onAccessibilityEvent(e: AccessibilityEvent?) {
        if (System.currentTimeMillis() > AutoSend.until) return
        val root = rootInActiveWindow ?: return
        val pkg = root.packageName?.toString() ?: return
        if (!pkg.startsWith("com.whatsapp")) return
        val cands = root.findAccessibilityNodeInfosByViewId(pkg + ":id/send") +
            root.findAccessibilityNodeInfosByText("Send")
        val btn = cands.firstOrNull { n ->
            val d = n.contentDescription?.toString() ?: ""
            d.contains("send", true) || d.contains("भेजें")
        } ?: return
        var n: AccessibilityNodeInfo? = btn
        while (n != null && !n.isClickable) n = n.parent
        if (n != null && n.performAction(AccessibilityNodeInfo.ACTION_CLICK)) {
            AutoSend.until = 0L
        }
    }

    override fun onInterrupt() {}
}
